# !/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
prepare_requirements.py

Генерирует requirements.txt и requirements-optional.txt для photo_processor_combined_auto_ai.py
"""
from pathlib import Path
import sys

CORE = [
    "numpy>=1.21",
    "opencv-python>=4.5",
    "Pillow>=8.0",
]

OPTIONAL = [
    "onnxruntime>=1.10   # optional: ONNX runtime for onnx models",
    "rembg>=2.0.32      # optional: background removal (may require pillow, numpy)",
    "streamlit>=1.0     # optional: UI",
    "transformers>=4.25  # optional: SAM via transformers (large)",
    # transformers SAM typically requires a deep-learning backend (torch). Install torch according to your CUDA:
    "torch>=1.12.0      # optional: install CPU or CUDA build manually, see https://pytorch.org/",
    "torchvision>=0.13  # optional companion to torch",
]

NOTES = [
    "# Notes:",
    "# - torch must be installed with the correct CUDA/CPU wheel for your system. See https://pytorch.org/get-started/locally/",
    "# - transformers + SAM model (facebook/sam-vit-huge) is very large (~>7GB model) and requires adequate RAM/GPU.",
    "# - rembg may require additional native dependencies (libwebp, etc.) depending on platform.",
]

def write_file(path: Path, lines):
    path.write_text("\n".join(lines) + "\n", encoding="utf-8")
    print(f"Wrote {path}")

def main():
    base = Path(".")
    write_file(base / "requirements.txt", CORE + [""] + ["# Optional extras are listed in requirements-optional.txt"])
    write_file(base / "requirements-optional.txt", OPTIONAL + [""] + NOTES)
    # helpful install instructions
    install_sh = base / "install_requirements.sh"
    install_sh.write_text(
        "#!/bin/sh\n"
        "set -e\n"
        "python -m pip install --upgrade pip\n"
        "python -m pip install -r requirements.txt\n"
        "echo \"To install optional extras, run: python -m pip install -r requirements-optional.txt\"\n",
        encoding="utf-8"
    )
    install_sh.chmod(0o755)
    print("Created install_requirements.sh - run it to install core deps.")
    print("Remember to install torch with the correct CUDA/CPU variant if you need SAM/transformers support.")

if __name__ == "__main__":
    main()
